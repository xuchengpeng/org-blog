#+TITLE: 使用 curl 探测下载
#+AUTHOR: Chuck
#+DESCRIPTION: 尝试从不同的地址下载文件，下载前先探测文件是否存在。
#+FILETAGS: :Linux:
#+DATE: <2026-11-24 Tue 13:54>

在 Linux 中经常使用 curl 从其他地方下载文件，有些文件可以尝试从多个服务器下载，可以先进行检测，根据结果判定当前可以从哪个地址进行下载。

检测时使用命令 =curl -fsIL "${url}" > /dev/null 2>&1= ，其中参数 =-f= 表示存在 HTTP 错误时快速失败返回，参数 =-I= 表示只显示文件头信息不会下载整个文件，参数 =-L= 表示跟随重定向（如 http→https 跳转），参数 =-s= 表示静默模式不显示进度条和错误信息。文件存在时（HTTP 200 OK）命令返回 0 ，文件不存在时（HTTP 404 等）命令返回非 0 。

下面的命令用于下载 Emacs 的代码包文件，先用正式发布版地址尝试下载，再从测试版地址尝试下载。

#+begin_src shell
  version="31.1" # 当前测试版的版本号 31.1.90
  ftp_url="https://ftp.gnu.org/gnu/emacs/emacs-${version}.tar.xz"
  alpha_url="https://alpha.gnu.org/gnu/emacs/pretest/emacs-${version}.tar.xz"

  if curl -fsIL "${ftp_url}" > /dev/null 2>&1; then
      url="${ftp_url}"
  elif curl -fsIL "${alpha_url}" > /dev/null 2>&1; then
      url="${alpha_url}"
  else
      echo "错误: 两个地址都找不到 emacs-${version}.tar.xz" >&2
      exit 1
  fi

  echo "从 ${url} 下载..."
  curl -fL -o "emacs-${version}.tar.xz" "${url}"
#+end_src
