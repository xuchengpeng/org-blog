#+TITLE: Build OpenCV From Source With mingw-w64
#+AUTHOR: Chuck
#+DESCRIPTION: A guide to build OpenCV on Windows with mingw-w64 toolchains.
#+FILETAGS: :ComputerVision:OpenCV:Windows:
#+DATE: <2025-12-04 Thu 13:21>

OpenCV (Open Source Computer Vision Library: http://opencv.org) is an open-source library that includes several hundreds of computer vision algorithms.

#+begin_src shell
git clone --depth 1 --branch 5.1.0 https://github.com/opencv/opencv.git
cd opencv
cmake -G "MinGW Makefiles" -DCMAKE_BUILD_TYPE=Release -DCMAKE_INSTALL_PREFIX=build/install -DBUILD_TESTS=OFF -DBUILD_PERF_TESTS=OFF -DBUILD_EXAMPLES=OFF -DBUILD_opencv_apps=OFF -DBUILD_JAVA=OFF -DBUILD_FAT_JAVA_LIB=OFF -DBUILD_opencv_python2=OFF -DBUILD_opencv_python3=OFF -DWITH_LAPACK=OFF -B build
cmake --build build -j 8
cmake --install build
#+end_src
